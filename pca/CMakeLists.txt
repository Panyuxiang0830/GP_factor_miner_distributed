cmake_minimum_required(VERSION 3.18)
project(pca LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(PCA_MPI "Enable MPI" ON)
option(PCA_CUDA "Enable hand-written CUDA kernels" OFF)
option(PCA_CUBLAS_REFERENCE "Build isolated cuBLAS comparison executable only" OFF)
option(PCA_ENGINE_BENCH "Build real-panel size benchmark using the factor engine" OFF)
if(PCA_CUBLAS_REFERENCE AND NOT PCA_CUDA)
  message(FATAL_ERROR "PCA_CUBLAS_REFERENCE requires PCA_CUDA")
endif()
find_package(OpenMP REQUIRED)
add_library(pca_core core.cpp ../src/core/json.cpp)
set_target_properties(pca_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(pca_core PUBLIC . ../include)
target_link_libraries(pca_core PUBLIC OpenMP::OpenMP_CXX)
add_executable(pca_run main.cpp)
target_link_libraries(pca_run PRIVATE pca_core)
add_library(pca_api SHARED c_api.cpp)
target_link_libraries(pca_api PRIVATE pca_core)
if(PCA_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set(CMAKE_CUDA_STANDARD 17)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  add_library(pca_gpu gpu.cu)
  set_target_properties(pca_gpu PROPERTIES CUDA_ARCHITECTURES 86)
  target_link_libraries(pca_gpu PUBLIC pca_core CUDA::cudart)
  target_link_libraries(pca_run PRIVATE pca_gpu)
  target_compile_definitions(pca_run PRIVATE PCA_CUDA)
  add_executable(pca_gpu_test test_gpu.cpp)
  target_link_libraries(pca_gpu_test PRIVATE pca_gpu)
  add_executable(pca_gpu_bench gpu_bench.cpp)
  target_link_libraries(pca_gpu_bench PRIVATE pca_gpu)
  if(PCA_CUBLAS_REFERENCE)
    add_executable(pca_cublas_reference reference_cublas.cu)
    set_target_properties(pca_cublas_reference PROPERTIES CUDA_ARCHITECTURES 86)
    target_link_libraries(pca_cublas_reference PRIVATE pca_core CUDA::cudart CUDA::cublas)
  endif()
endif()
if(PCA_MPI)
  find_package(MPI REQUIRED)
  target_link_libraries(pca_run PRIVATE MPI::MPI_CXX)
  target_compile_definitions(pca_run PRIVATE PCA_MPI)
  add_executable(pca_archive_test test_mpi_archive.cpp)
  target_link_libraries(pca_archive_test PRIVATE pca_core MPI::MPI_CXX)
  add_executable(pca_archive_bench archive_bench.cpp)
  target_link_libraries(pca_archive_bench PRIVATE pca_core MPI::MPI_CXX)
  add_executable(pca_matched_bench matched_bench.cpp)
  target_link_libraries(pca_matched_bench PRIVATE pca_core MPI::MPI_CXX)
endif()
if(PCA_ENGINE_BENCH)
  add_library(pca_bench_engine STATIC
    ../src/core/dag.cpp ../src/core/panel_io.cpp ../src/core/ops.cpp
    ../src/core/operators.cpp ../src/core/evaluate.cpp
    ../src/core/metrics.cpp ../src/core/ema_scan.cpp ../src/core/halo_plan.cpp)
  target_link_libraries(pca_bench_engine PUBLIC pca_core)
  if(PCA_MPI)
    add_library(pca_engine_mpi STATIC ../src/mpi/distributed.cpp)
    target_link_libraries(pca_engine_mpi PUBLIC pca_bench_engine MPI::MPI_CXX)
    add_executable(pca_engine_bridge engine_bridge.cpp)
    target_link_libraries(pca_engine_bridge PRIVATE pca_engine_mpi)
    if(PCA_CUDA)
      target_link_libraries(pca_engine_bridge PRIVATE pca_gpu)
      target_compile_definitions(pca_engine_bridge PRIVATE PCA_CUDA)
    endif()
  endif()
  add_executable(pca_real_size_bench real_size_bench.cpp)
  target_link_libraries(pca_real_size_bench PRIVATE pca_bench_engine)
  if(PCA_CUDA)
    add_executable(pca_rankic_test test_rankic_interface.cpp)
    target_link_libraries(pca_rankic_test PRIVATE pca_bench_engine pca_gpu)
  endif()
endif()
enable_testing()
add_executable(pca_test test_core.cpp)
target_link_libraries(pca_test PRIVATE pca_core)
add_test(NAME pca_core COMMAND pca_test)
